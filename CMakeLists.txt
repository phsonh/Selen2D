cmake_minimum_required(VERSION 3.25)
# This is the only maintained product-version value. Runtime, Windows resources
# and distributable metadata are generated from it; revisions are developmental.
project(Engine VERSION 2.0.0 LANGUAGES C CXX RC)
# Distributable binaries use the static MSVC runtime, including all dependencies.
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(ENGINE_PRODUCT_NAME "Selen2D")
set(ENGINE_DEVELOPER "Lonely Owl")
set(ENGINE_COPYRIGHT "Copyright (c) 2026 ${ENGINE_DEVELOPER}")
set(ENGINE_FILE_VERSION "${PROJECT_VERSION}.0")
set(ENGINE_RUNTIME_NAME "${ENGINE_PRODUCT_NAME}")
set(ENGINE_PACK_NAME "${ENGINE_PRODUCT_NAME}Pack")
configure_file(src/engine/BuildInfo.h.in generated/BuildInfo.h @ONLY)
configure_file(tools/product-info.json.in bin/product-info.json @ONLY)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
option(BUILD_TESTING "Build local tests (not included in the source distribution)" OFF)
if(BUILD_TESTING AND NOT EXISTS "${PROJECT_SOURCE_DIR}/tests/CMakeLists.txt")
    message(FATAL_ERROR "BUILD_TESTING requires local tests/CMakeLists.txt. Tests are not included in this checkout; configure with -DBUILD_TESTING=OFF for the engine build.")
endif()
find_package(Threads REQUIRED)

add_subdirectory(3rd_party)
include(tools/DxcStatic.cmake)
engine_add_static_dxc()
add_subdirectory(src)
if(BUILD_TESTING)
    include(CTest)
    add_subdirectory(tests)
endif()
include(tools/StageRelease.cmake)
engine_stage_release()
